% Subtitle handling — store separately for professional title page
\makeatletter
\newcommand{\@docsubtitle}{}
\def\subtitle#1{\renewcommand{\@docsubtitle}{#1}}
\newcommand{\@docversionrows}{}
\newcommand{\addversionhistory}[3]{%
  \g@addto@macro\@docversionrows{#1 & #2 & #3 \\}%
}

% Professional title page
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \begin{tikzpicture}[remember picture, overlay]
      % Top colored band
      \fill[blue!70!black] (current page.north west) rectangle ([yshift=-5cm]current page.north east);
      % Accent line below top band
      \fill[blue!40!white] ([yshift=-5cm]current page.north west) rectangle ([yshift=-5.15cm]current page.north east);
      % Bottom colored band
      \fill[blue!70!black] (current page.south west) rectangle ([yshift=1.5cm]current page.south east);
    \end{tikzpicture}
    \vspace*{5cm}
    \begin{center}
      {\fontsize{28}{36}\selectfont\bfseries\color{blue!70!black} \@title \par}
      \vspace{1cm}
      {\Large\color{gray!60!black} \@docsubtitle \par}
      \vspace{2cm}
      {\large\color{gray!50!black} \@date \par}
      \ifx\@docversionrows\@empty\else
        \vspace{1.4cm}
        {\small
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{p{1.2cm}p{3.3cm}p{8.1cm}}
          \toprule
          \textbf{Version} & \textbf{Date} & \textbf{Change summary} \\
          \midrule
          \@docversionrows
          \bottomrule
        \end{tabular}
        \par}
      \fi
    \end{center}
    \vfill
  \end{titlepage}
}
\makeatother

% Per-document option banners and matching section colors
\newcommand{\optionbanner}[3]{%
  \definecolor{CurrentOptionColor}{HTML}{#1}%
  \colorlet{DocSectionColor}{CurrentOptionColor}%
  \colorlet{DocSubsectionColor}{CurrentOptionColor}%
  \colorlet{DocHeaderColor}{CurrentOptionColor}%
  \renewcommand{\DocHeadRuleWidth}{1.2pt}%
  \vspace{0.2em}
  \noindent\colorbox{CurrentOptionColor}{%
    \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
      \vspace{0.22cm}%
      {\color{white}\LARGE\bfseries #2\par}%
      \vspace{0.08cm}%
      {\color{white}\large #3\par}%
      \vspace{0.22cm}%
    }%
  }\par\vspace{0.5em}
}

\newcommand{\optionlegenditem}[3]{%
  \definecolor{CurrentLegendColor}{HTML}{#1}%
  \noindent\colorbox{CurrentLegendColor}{%
    \makebox[1.4cm][c]{\color{white}\bfseries #2}%
  }%
  \hspace{0.6em}\textbf{#3}\par\vspace{0.18em}%
}

\newcommand{\methodbanner}[3]{%
  \resetsectioncolor
  \definecolor{CurrentMethodColor}{HTML}{#1}%
  \vspace{0.2em}
  \noindent
  \begingroup
    \colorbox{CurrentMethodColor}{%
      \parbox[c][1.15cm][c]{\dimexpr\linewidth-2\fboxsep\relax}{%
        \hspace{0.25cm}%
        {\color{white}\bfseries #2}%
        \hspace{0.45cm}%
        {\color{white}\large\bfseries #3}%
      }%
    }%
  \endgroup
  \par\vspace{0.45em}
}
